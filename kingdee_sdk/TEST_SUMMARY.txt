from k3cloud_webapi_sdk.main import K3CloudApiSdk

# 创建SDK实例
api_sdk = K3CloudApiSdk("http://k3.forestpacking.com/k3cloud", timeout=420)

# 初始化配置
api_sdk.InitConfig(
    acct_id="66306281bdfb20",
    user_name="Administrator", 
    app_id="337602_241BWwjKyODY5bxGWZ0t3/XqyJ28xBOt",
    app_secret="YOUR_APP_SECRET_HERE",  # <-- 需要替换
    server_url="http://k3.forestpacking.com/k3cloud",
    lcid=2052,
    org_num=0,
    connect_timeout=120,
    request_timeout=420
)

# 查询物料示例
para = {
    "FormId": "BD_MATERIAL",
    "FieldKeys": "FMaterialId,FNumber,FName",
    "FilterString": "FName like '%原纸%'",
    "Limit": 100
}
response = api_sdk.ExecuteBillQuery(para)
print(response)
